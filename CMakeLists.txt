cmake_minimum_required(VERSION 3.16)
project(fast_modbus LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(FMB_BUILD_DRIVER "Build the MasterPLC addin (requires vendor SDK)" OFF)
option(FMB_SANITIZERS "Address/undefined sanitizers for host tests" OFF)
set(SRC ${CMAKE_CURRENT_SOURCE_DIR}/mplc_protocol_fast_modbus)
add_library(fmb_core STATIC ${SRC}/fmb_core.cpp ${SRC}/fmb_frames.cpp ${SRC}/fmb_transport.cpp)
target_include_directories(fmb_core PUBLIC ${SRC})
set_target_properties(fmb_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(WIN32)
  target_compile_definitions(fmb_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(fmb_core PUBLIC ws2_32)
else()
  target_link_libraries(fmb_core PUBLIC pthread)
  target_compile_options(fmb_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(FMB_SANITIZERS AND NOT MSVC)
  target_compile_options(fmb_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(fmb_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE fmb_core)
enable_testing()
add_test(NAME core COMMAND core_tests)
if(FMB_BUILD_DRIVER)
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "arm")
    message(FATAL_ERROR "For ARM addins use tools/build_arm_vendor.py: vendor C++ ABI and pinned runtime libraries are required")
  endif()
  if(NOT IS_DIRECTORY "${MASTERPLC_SDK}/include")
    message(FATAL_ERROR "MASTERPLC_SDK must point to the target-compatible vendor API tree")
  endif()
  add_library(mplc_protocol_fast_modbus SHARED
    ${SRC}/fast_modbus_protocol.cpp ${SRC}/fast_modbus_module.cpp
    ${SRC}/fast_modbus_channel.cpp ${SRC}/mplc_fast_modbus.cpp)
  set_property(TARGET mplc_protocol_fast_modbus PROPERTY CXX_STANDARD 20)
  target_link_libraries(mplc_protocol_fast_modbus PRIVATE fmb_core)
  target_include_directories(mplc_protocol_fast_modbus SYSTEM PRIVATE
    ${MASTERPLC_SDK}/include ${MASTERPLC_SDK}/include/core ${MASTERPLC_SDK}/lib ${MASTERPLC_SDK}/lib/opcua)
  target_compile_definitions(mplc_protocol_fast_modbus PRIVATE
    MPLC_ADDIN_EXPORTS MPLC_TPWM_EXPORTS BOOST_USER_CONFIG="mplc/detail/mplc_boost_config.hpp")
  if(WIN32)
    target_link_directories(mplc_protocol_fast_modbus PRIVATE ${MASTERPLC_LIBDIR})
    target_link_libraries(mplc_protocol_fast_modbus PRIVATE opcua mplcshare masterplc lua)
  else()
    target_compile_definitions(mplc_protocol_fast_modbus PRIVATE LINUX _GNUC_ _DEFAULT_SOURCE TARGET_PLATFORM_NAME="linux-x64")
    foreach(name masterplc mplcshare opcua liblua)
      if(NOT EXISTS "${MASTERPLC_LIBDIR}/${name}.so")
        message(FATAL_ERROR "Missing verified target library: ${MASTERPLC_LIBDIR}/${name}.so")
      endif()
      target_link_libraries(mplc_protocol_fast_modbus PRIVATE "${MASTERPLC_LIBDIR}/${name}.so")
    endforeach()
    target_link_libraries(mplc_protocol_fast_modbus PRIVATE dl rt)
    set_target_properties(mplc_protocol_fast_modbus PROPERTIES PREFIX "" BUILD_RPATH "\$ORIGIN" INSTALL_RPATH "\$ORIGIN")
    target_link_options(mplc_protocol_fast_modbus PRIVATE -Wl,--no-undefined)
  endif()
endif()
